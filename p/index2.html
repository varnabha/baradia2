CREATE DATABASE SalesDB;
USE SalesDB;

CREATE TABLE Sales (
    sale_id INT PRIMARY KEY,
    product_name VARCHAR(100),
    product_price INT,
    quantity INT,
    sale_date DATE
);

DELIMITER //
CREATE PROCEDURE add_sale(
    IN sid INT,
    IN pname VARCHAR(25),
    IN price INT,
    IN qty INT,
    IN sdate DATE
)
BEGIN
    INSERT INTO Sales VALUES (sid, pname, price, qty, sdate);
END//

DELIMITER //
CREATE FUNCTION get_total_sales()
RETURNS INT
DETERMINISTIC
BEGIN
    DECLARE total INT;
    SELECT SUM(product_price * quantity) INTO total FROM Sales;
    RETURN total;
END//
DELIMITER ;

SELECT get_total_sales() AS total_sales;

DELIMITER //
CREATE TRIGGER after_employee_insert
AFTER INSERT ON employees
FOR EACH ROW
BEGIN
    INSERT INTO employees_audit VALUES
    (NEW.Emp_ID, NEW.Emp_Name, NEW.Salary, NEW.Dept_Name);
END//
DELIMITER ;

DELIMITER //
CREATE PROCEDURE update_it_salary()
BEGIN
    DECLARE done INT DEFAULT 0;
    DECLARE e_id INT;
    DECLARE e_sal INT;
    DECLARE cur CURSOR FOR
        SELECT Emp_ID, Salary FROM employees WHERE Dept_Name = 'IT';
    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = 1;
    OPEN cur;
    sal_loop: LOOP
        FETCH cur INTO e_id, e_sal;
        IF done = 1 THEN
            LEAVE sal_loop;
        END IF;
        UPDATE employees SET Salary = e_sal * 1.10 WHERE Emp_ID = e_id;
    END LOOP;
    CLOSE cur;
END//
DELIMITER ;

SET SQL_SAFE_UPDATES = 0;

SELECT Emp_ID, Emp_Name, Salary FROM employees WHERE Dept_Name = 'IT';

CALL update_it_salary();

SELECT Emp_ID, Emp_Name, Salary FROM employees WHERE Dept_Name = 'IT';

▼
CMD — Step 1: Navigate 
cd "C:\Program Files\MySQL\MySQL Server 8.0\bin"
CMD — Step 2: Take Backup 
mysqldump -u root -p SchoolDB > C:\SchoolDB_backup.sql
SQL — Step 3: Simulate Data Loss 
DROP DATABASE SchoolDB;
SHOW DATABASES;
CMD — Step 4: Restore 
mysql -u root -p -e "CREATE DATABASE SchoolDB;"
mysql -u root -p SchoolDB < C:\SchoolDB_backup.sql

USE SchoolDB;

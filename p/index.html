# Vacuum Cleaner Agent

rooms = {'A': 'Dirty', 'B': 'Dirty'}
location = 'A'

while 'Dirty' in rooms.values():
    print("Location:", location)

    if rooms[location] == 'Dirty':
        print("Action: Suck")
        rooms[location] = 'Clean'
    else:
        print("Action: Move")

    location = 'B' if location == 'A' else 'A'

3. Practical 3: Tower of Hanoi

def hanoi(n, source, helper, destination):
    if n == 1:
        print(source, "to", destination)
    else:
        hanoi(n-1, source, destination, helper)
        print(source, "to", destination)
        hanoi(n-1, helper, source, destination)

4. DFS — Water Jug Problem

def dfs(a, b, seen):
    if (a, b) in seen:
        return False
    seen.add((a, b))
    print(a, b)

    if a == 2:
        return True

    states = [
        (4, b), (a, 3), (0, b), (a, 0),
        (a-min(a, 3-b), b+min(a, 3-b)),
        (a+min(b, 4-a), b-min(b, 4-a))
    ]

    for x, y in states:
        if dfs(x, y, seen):
            return True
    return False

5. Practical: Breadth-First Search (BFS)

from collections import deque

graph = {'A': ['B', 'C'], 'B': ['D'], 'C': ['E'], 'D': [], 'E': []}

q = deque(['A'])
visited = ['A']

while q:
    node = q.popleft()
    print(node, end=" ")

    for x in graph[node]:
        if x not in visited:
            visited.append(x)
            q.append(x)

6. 8-Queens using Backtracking

N = 8
board = []

def solve(row):
    if row == N:
        print(board)
        return True

    for col in range(N):
        safe = True

        for r in range(row):
            if board[r] == col:
                safe = False
                break
            if abs(board[r] - col) == row - r:
                safe = False
                break

        if safe:
            board.append(col)
            if solve(row + 1):
                return True
            board.pop()

    return False

solve(0)

7. Local Beam Search for Travelling Salesman Problem (TSP)

import random

cities = [0, 1, 2, 3]
dist = [[0, 10, 15, 20],
        [10, 0, 35, 25],
        [15, 35, 0, 30],
        [20, 25, 30, 0]]

beam = [[0, 1, 2, 3], [0, 2, 1, 3]]

for _ in range(5):
    new = []
    for path in beam:
        for i in range(1, len(path)-1):
            p = path[:]
            p[i], p[i+1] = p[i+1], p[i]
            new.append(p)

    def cost(p):
        return sum(dist[p[i]][p[i+1]] for i in range(3)) + dist[p[-1]][p[0]]

    beam = sorted(new, key=cost)[:2]

best = min(beam, key=cost)
print("Best Path:", best)
print("Minimum Cost:", cost(best))

8. A* Algorithm for 8-Puzzle Problem

from heapq import heappush, heappop

start = (1, 2, 3, 4, 0, 6, 7, 5, 8)
goal = (1, 2, 3, 4, 5, 6, 7, 8, 0)

def h(s):
    return sum(s[i] != 0 and s[i] != goal[i] for i in range(9))

q = [(h(start), 0, start)]
visited = set()

while q:
    f, g, s = heappop(q)

    if s == goal:
        print("Goal Reached:", s)
        break

    if s in visited:
        continue
    visited.add(s)

    z = s.index(0)

    for d in [-3, 3, -1, 1]:
        n = z + d
        if 0 <= n < 9 and not (d == -1 and z % 3 == 0) and not (d == 1 and z % 3 == 2):
            t = list(s)
            t[z], t[n] = t[n], t[z]
            t = tuple(t)

            if t not in visited:
                heappush(q, (g + 1 + h(t), g + 1, t))
